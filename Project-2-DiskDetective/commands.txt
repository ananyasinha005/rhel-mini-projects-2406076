===== DISKDETECTIVE COMMANDS =====

# Check storage layout
lsblk

# Check filesystem usage
df -h

# Check directory usage
sudo du -sh /home/* /var/* /opt/* 2>/dev/null | sort -rh

# Find files larger than 100 MB
sudo find / -type f -size +100M 2>/dev/null

# Find large files on the main filesystem
sudo find / -xdev -type f -size +100M -printf '%s %p\n' 2>/dev/null | sort -nr

# Find files older than 180 days
sudo find / -xdev -type f -mtime +180 2>/dev/null

# Find first 20 stale files
sudo find / -xdev -type f -mtime +180 2>/dev/null | head -20

# Find files owned by dev1
sudo find / -xdev -type f -user dev1 2>/dev/null

# Create test directories
sudo mkdir -p /data/renders /data/footage

# Create 150 MB test file
sudo dd if=/dev/zero of=/data/renders/final_cut_v3.mp4 bs=1M count=150

# Create 200 MB test file
sudo dd if=/dev/zero of=/data/footage/raw_footage_2024.mov bs=1M count=200

# Create stale files
sudo touch -d "300 days ago" /data/footage/old_project_backup.mov
sudo touch -d "250 days ago" /data/renders/abandoned_edit.mp4
sudo touch -d "10 days ago" /data/renders/recent_work.mp4

# Create file owned by dev1
sudo touch /data/footage/dev1_project.mov
sudo chown dev1 /data/footage/dev1_project.mov

# Verify test data
ls -lh /data/renders /data/footage

# Large-file audit
sudo find /data -type f -size +100M 2>/dev/null

# Stale-file audit
sudo find /data -type f -mtime +180 2>/dev/null

# Ownership audit
sudo find /data -type f -user dev1 2>/dev/null

# Hard and symbolic links
echo "original content" | sudo tee /data/testfile.txt
sudo ln /data/testfile.txt /data/hardlink.txt
sudo ln -s /data/testfile.txt /data/symlink.txt
ls -li /data/testfile.txt /data/hardlink.txt /data/symlink.txt

# Delete original
sudo rm /data/testfile.txt

# Test hard link
cat /data/hardlink.txt

# Test symbolic link
cat /data/symlink.txt

# Redirection
sudo find /data -type f 2>/dev/null > /root/DiskDetective/file_list.txt

# Append using tee
date | sudo tee -a /root/DiskDetective/file_list.txt

# Sort, head and discard errors
sudo du -sh /data/* 2>/dev/null | sort -rh | head -10

# Count files
sudo find /data -type f 2>/dev/null | wc -l

# Calculate reclaimable space
sudo find /data -type f -mtime +180 -exec du -ch {} + 2>/dev/null | tail -1

# Add measurable data to stale file
sudo dd if=/dev/zero of=/data/footage/old_project_backup.mov bs=1M count=1
sudo touch -d "300 days ago" /data/footage/old_project_backup.mov

# Add second disk and verify
lsblk
sudo mkfs.ext4 /dev/nvme0n2
sudo mkdir -p /mnt/coldarchive
sudo mount /dev/nvme0n2 /mnt/coldarchive
df -h /mnt/coldarchive

# Copy stale files to cold archive
sudo find /data -type f -mtime +180 -exec cp {} /mnt/coldarchive/ \;

# Verify archive
ls -lh /mnt/coldarchive
mount | grep coldarchive

# Create storage audit report
sudo tee /root/DiskDetective/storage_audit.txt

# Verify report
sudo ls -lh /root/DiskDetective/storage_audit.txt
sudo cat /root/DiskDetective/storage_audit.txt
sudo head -15 /root/DiskDetective/storage_audit.txt
